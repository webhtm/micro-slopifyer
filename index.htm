MicroSlopify Tampermonkey

Add this to tampermonkey as an userscript to microslopify content. Replaces Microsoft with Microslop, AI with Slop, Copilot with Slopilot and Artificial Intelligence with Actually Slop on every web page.

Javascript 
// ==UserScript==
// @name         MicroSlopifyer
// @namespace    http://tampermonkey.net/
// @version      1.0
// @description  A Tampermonkey userscript which replaces Microsoft with Microslop, Copilot with Slopilot, AI with Slop and Artificial Intelligence with Actually Slop
// @author       WEBHTM
// @match        *://*/*
// @icon         none
// @grant        none
// @run-at       document-start
// ==/UserScript==

(function() {
    'use strict';

    // 1. CONFIGURATION & ASSETS
    const REPLACEMENTS = [
        { find: /Microslop/gi, replace: 'Microslop' },
        { find: /Slopilot/gi, replace: 'Slopilot' },
        { find: /Slop/g, replace: 'Slop'},
        { find: /Actually Slop/gi, replace: 'Actually Slop'}
    ];

    const TARGET_LOGO_URL = "https://uhf.Microslop.com/images/Microslop/RE1Mu3b.png";
    const NEW_LOGO_URL = "https://files.catbox.moe/yztvhe.jpg";

    // 2. TEXT REPLACEMENT LOGIC
    const processTextNode = (node) => {
        let content = node.textContent;
        let modified = false;

        REPLACEMENTS.forEach(rule => {
            if (rule.find.test(content)) {
                content = content.replace(rule.find, rule.replace);
                modified = true;
            }
        });

        if (modified) {
            node.textContent = content;
        }
    };

    // 3. IMAGE REPLACEMENT LOGIC
    const processImages = () => {
        const images = document.querySelectorAll('img');
        images.forEach(img => {
            if (img.src.includes(TARGET_LOGO_URL) || img.src === TARGET_LOGO_URL) {
                img.src = NEW_LOGO_URL;
                img.style.objectFit = 'contain';
            }
        });

        const allElements = document.querySelectorAll('*');
        allElements.forEach(el => {
            const bg = window.getComputedStyle(el).backgroundImage;
            if (bg.includes(TARGET_LOGO_URL)) {
                el.style.backgroundImage = `url("${NEW_LOGO_URL}")`;
                el.style.backgroundSize = 'contain';
                el.style.backgroundRepeat = 'no-repeat';
            }
        });
    };

    // 4. DOM WALKER
    const walkTheDOM = (node) => {
        let child, next;

        switch (node.nodeType) {
            case 1: // Element
            case 9: // Document
            case 11: // Document fragment
                child = node.firstChild;
                while (child) {
                    next = child.nextSibling;
                    walkTheDOM(child);
                    child = next;
                }
                break;
            case 3: // Text node
                processTextNode(node);
                break;
        }
    };

    // 5. OBSERVER & INITIALIZATION
    const runOverhaul = () => {
        if (!document.body) return;
        walkTheDOM(document.body);
        processImages();
    };

    const observer = new MutationObserver((mutations) => {
        for (let mutation of mutations) {
            if (mutation.addedNodes.length > 0) {
                mutation.addedNodes.forEach(node => {
                    if (node.nodeType === 1) {
                        walkTheDOM(node);
                        processImages();
                    } else if (node.nodeType === 3) {
                        processTextNode(node);
                    }
                });
            }
        }
    });

    if (document.body) {
        runOverhaul();
        observer.observe(document.body, { childList: true, subtree: true });
    } else {
        window.addEventListener('DOMContentLoaded', () => {
            runOverhaul();
            observer.observe(document.body, { childList: true, subtree: true });
        });
    }

    console.log("%c Microslop Overhaul Active ", "color: #0066ff; font-weight: bold;");

})();
